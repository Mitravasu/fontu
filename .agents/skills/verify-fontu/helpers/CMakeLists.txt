cmake_minimum_required(VERSION 3.21)
project(fontu_verification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Test)
add_subdirectory("${FONTU_SOURCE}" fontu EXCLUDE_FROM_ALL)
add_executable(fontu_ui_verify driver.cpp)
target_compile_definitions(fontu_ui_verify PRIVATE FONTU_MAIN="${FONTU_SOURCE}/src/main.cpp")
target_link_libraries(fontu_ui_verify PRIVATE fontu_canvas Qt6::Test)
