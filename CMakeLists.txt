cmake_minimum_required(VERSION 3.21)
project(fontu VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_library(fontu_core src/document.cpp src/shape_builder.cpp src/viewport.cpp src/snapping.cpp src/project.cpp src/project_creation.cpp src/font_description.cpp src/font_export.cpp)
target_include_directories(fontu_core PUBLIC src)
target_link_libraries(fontu_core PUBLIC Qt6::Gui)
add_library(fontu_canvas src/canvas.cpp src/reference_glyph.cpp src/snap_menu.cpp src/glyph_panel.cpp src/project_window.cpp src/creation_page.cpp)
target_link_libraries(fontu_canvas PUBLIC fontu_core Qt6::Widgets)
add_executable(fontu MACOSX_BUNDLE src/main.cpp)
target_link_libraries(fontu PRIVATE fontu_canvas)
set_target_properties(fontu PROPERTIES
    WIN32_EXECUTABLE TRUE
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.fontu.canvas"
    MACOSX_BUNDLE_BUNDLE_NAME "fontu"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
if(APPLE)
    set_target_properties(fontu PROPERTIES INSTALL_RPATH "@executable_path/../Frameworks")
endif()
install(TARGETS fontu BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES LICENSE DESTINATION share/licenses/fontu)
if(UNIX AND NOT APPLE)
    install(FILES packaging/fontu.desktop DESTINATION share/applications)
    install(FILES packaging/fontu.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/release-version.txt" CONTENT "${PROJECT_VERSION}\n")
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE fontu_core Qt6::Test)
    add_test(NAME core COMMAND core_tests)
    add_executable(project_tests tests/project_tests.cpp)
    target_link_libraries(project_tests PRIVATE fontu_core Qt6::Test)
    add_test(NAME project COMMAND project_tests -platform offscreen)
    add_executable(project_window_tests tests/project_window_tests.cpp)
    target_link_libraries(project_window_tests PRIVATE fontu_canvas Qt6::Test)
    add_test(NAME project_window COMMAND project_window_tests -platform offscreen)
    set_tests_properties(project_window PROPERTIES TIMEOUT 30)
    add_executable(font_export_tests tests/font_export_tests.cpp)
    target_link_libraries(font_export_tests PRIVATE fontu_core Qt6::Test)
    add_test(NAME font_export COMMAND font_export_tests -platform offscreen)
    add_executable(canvas_tests tests/canvas_tests.cpp)
    target_link_libraries(canvas_tests PRIVATE fontu_canvas Qt6::Test)
    add_test(NAME canvas COMMAND canvas_tests -platform offscreen)
endif()
